// Conectarse a mongoDB
mongosh

//ver bases de datos
show dbs

//entrar en una base de datos
use admin

//crear y entrar en una base de datos
use db_carros

//ver la base de datos en la cual se encuentra el sistema
db

//borrar base de datos
db.dropDatabase()


//crear nueva coleccion
 db.createCollection("carros")

//ver coleccion
show collections

//borrar coleccion
db.carros.drop()

//insertar un documento JSON
db.carros.insertOne(
    {
        "placa":"vem072",
        "numero_serie": 12000384,
        "modelo": 2024,
        "marca" : "Chevrolet",
        "kilometraje" : 23000,
        "tipo": "sedan"
    }
)

//ver todos los documentos dentro de una coleccion
db.carros.find()

//ver un documento en especifico
db.carros.find( 
    {
        "placa" : "vem072"
    }
)
